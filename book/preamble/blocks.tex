% Distinct statement blocks. Shared counter: Definition 1.1, Theorem 1.2, ...
% Proof is unnumbered. Colours come from the active theme.

\renewcommand{\qedsymbol}{\textcolor{accent}{$\blacksquare$}}

\tcbset{
  moiblockbase/.style={
    enhanced jigsaw,
    breakable,
    sharp corners,
    boxrule=0pt,
    coltext=bodyfg,
    colback=#1,
    left=10pt,
    right=10pt,
    top=6pt,
    bottom=7pt,
    before skip=12pt,
    after skip=12pt,
  }
}

% Definition: warm fill, solid left bar, small diamond.
\tcolorboxenvironment{definition}{
  moiblockbase=defbg,
  colframe=defbar,
  leftrule=3pt,
  overlay={
    \fill[defbar] ([xshift=5.5pt,yshift=-11pt]frame.north west)
      +(0,2.1pt) -- +(2.1pt,0) -- +(0,-2.1pt) -- +(-2.1pt,0) -- cycle;
  }
}

% Theorem: deeper fill, L-shaped top+left rule, small square.
\tcolorboxenvironment{theorem}{
  moiblockbase=thmbg,
  colframe=thmbar,
  leftrule=3pt,
  toprule=1.35pt,
  overlay={
    \fill[thmbar] ([xshift=4.4pt,yshift=-10.6pt]frame.north west)
      rectangle ++(3.1pt,-3.1pt);
  }
}

% Corollary: lighter fill, dashed left bar, open square.
\tcolorboxenvironment{corollary}{
  moiblockbase=corbg,
  colframe=pagebg,
  borderline west={1.35pt}{0pt}{corbar,dashed},
  overlay={
    \draw[corbar,line width=0.7pt]
      ([xshift=4.6pt,yshift=-10.4pt]frame.north west)
      rectangle ++(2.9pt,-2.9pt);
  }
}

% Proof: no fill, hairline, unnumbered.
\tcolorboxenvironment{proof}{
  moiblockbase=pagebg,
  colframe=prfbar,
  leftrule=0.55pt,
}

% Example: olive family, solid left bar, small circle.
\tcolorboxenvironment{example}{
  moiblockbase=exbg,
  colframe=exbar,
  leftrule=3pt,
  overlay={
    \fill[exbar] ([xshift=6pt,yshift=-12pt]frame.north west) circle (1.55pt);
  }
}

% Quiet note. Unnumbered.
\tcolorboxenvironment{remark}{
  moiblockbase=rembg,
  colframe=mutedfg,
  leftrule=0.7pt,
}

\tcolorboxenvironment{proposition}{
  moiblockbase=thmbg,
  colframe=thmbar,
  leftrule=3pt,
  toprule=1.35pt,
}

\tcolorboxenvironment{lemma}{
  moiblockbase=thmbg,
  colframe=thmbar,
  leftrule=3pt,
  toprule=1.35pt,
}

\tcolorboxenvironment{assumption}{
  moiblockbase=defbg,
  colframe=defbar,
  leftrule=3pt,
}
